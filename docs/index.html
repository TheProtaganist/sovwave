Quick Start Guide

Install Sovwave and run your first continuous wave training loop:

# 1. Install via Julia Pkg
using Pkg
Pkg.add(url="https://github.com/TheProtaganist/sovwave.git")

# 2. Train and Serialize Model
using Sovwave
using Sovwave.Audio

cfg = load_config("model_config.yaml")
field = create_field(cfg.field)
model = WaveModel(cfg)

inputs = [rand(cfg.model.embed_dims) for _ in 1:100]
targets = [rand(cfg.model.embed_dims) for _ in 1:100]

# Train via wave evolution seeking ground state
train!(model, inputs, targets, cfg.train)

# Save as dual-stream MKV video with presentation audio!
save_model(model, "wave_brain.mkv"; video_cfg=cfg.video, audio_cfg=cfg.audio)

# Run direct inference from the video file
output = infer("wave_brain.mkv", inputs[1])
